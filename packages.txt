# Paquetes que necesita Reddots, uno por línea (lo que va tras # es un comentario).
# Los instala update.sh, también en cada actualización: si falta alguno (porque lo has
# desinstalado o porque es nuevo en el repo), vuelve. Solo deben ir aquí los que usa la
# config (barra, atajos, capturas...); las apps que se usan sin más van en packages_opt.txt.
# Van con paru, así que pueden ser de los repos o de AUR.
quickshell      # El gestor de widgets, barra, etc
alacritty       # La terminal: la de Super + T y la que lanza la barra (install.sh quita kitty)
cmatrix         # El salvapantallas del menú de apagado (Power.qml): caracteres cayendo estilo "Matrix"

# Usados por los atajos de teclado (hypr/keybinds.lua)
wireplumber     # Trae wpctl, para las teclas de volumen y silenciar
brightnessctl   # Para las teclas de brillo de la pantalla
ddcutil         # Brillo de los monitores externos por DDC/CI (sliders de brillo del selector de tema, ThemeSettings.qml)
playerctl       # Para las teclas multimedia (siguiente, anterior, pausa)

# Usados por la barra (quickshell/). CachyOS ya trae alguno, pero así no dependo de eso
ttf-cascadia-code-nerd  # Fuente de la que salen los iconos de la barra (Nerd Font)
papirus-icon-theme      # Tema de iconos de Quickshell: lanzador, notificaciones, bandeja (pragma IconTheme en shell.qml)
libnotify               # Trae notify-send, para los avisos de la barra (batería baja, Bluetooth, wifi) y de las capturas
libpulse                # Trae pactl, para saber qué salidas de audio tienen algo enchufado (Volume.qml)
xdg-utils               # Trae xdg-open, para el botón "Abrir carpeta" de los fondos de pantalla
git                     # Para los botones de actualizar e instalar Reddots (scripts/update-reddots.sh)
pacman-contrib          # Trae checkupdates, para el contador de actualizaciones pendientes (Updates.qml)
cliphist                # Historial del portapapeles (Clipboard.qml; lo arranca hypr/hyprland.lua con wl-paste de wl-clipboard)
networkmanager          # Redes wifi y cable del widget de red (Network.qml: Quickshell.Networking habla con NetworkManager)
libpsl                  # Trae psl, para sacar el dominio del usuario en las redes wifi empresariales como eduroam (scripts/wifi-eap-connect.sh)
bluez                   # El servicio de Bluetooth del que tira Bluetooths.qml (Quickshell.Bluetooth)
upower                  # Datos de la batería (Battery.qml, BatteryMonitor.qml y Bar.qml: Quickshell.Services.UPower)
btop                    # Monitor de recursos en terminal

# Del sistema, no de la barra
xdg-desktop-portal-hyprland # Para compartir pantalla (Teams, Chrome...) y para las apps que piden hacer capturas

# Capturas de pantalla (Screenshot.qml y scripts/screenshot.sh)
grim            # Hace la captura
slurp           # Para elegir con el ratón la región o la ventana a capturar
hyprpicker      # Congela la pantalla mientras se elige, para poder capturar menús
wl-clipboard    # Trae wl-copy, para copiar la captura al portapapeles
jq              # Para leer las ventanas y monitores que da hyprctl en JSON
xdg-user-dirs   # Trae xdg-user-dir, para guardar en la carpeta de imágenes (~/Imágenes)
